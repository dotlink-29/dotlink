        // 动态调整画布大小（响应式设计） - 修复棋盘显示问题
        function resizeCanvas() {
            const container = document.querySelector('.grid-container');
            const containerWidth = container.clientWidth;
            
            // 计算合适的格子大小
            const maxGridWidth = gameState.gridSize.width;
            const maxGridHeight = gameState.gridSize.height;
            
            // 根据设备类型调整格子大小
            let baseCellSize;
            const isMobile = window.innerWidth <= 768;
            
            if (isMobile) {
                // 手机和平板：确保棋盘适应屏幕
                baseCellSize = Math.max(15, Math.min(22, (containerWidth - 40) / (maxGridWidth + 2)));
            } else {
                // 桌面：使用更大的格子，但确保不会超出容器
                // 计算最大可用宽度和高度
                const availableWidth = containerWidth - 40;
                const availableHeight = window.innerHeight * 0.6; // 使用屏幕高度的60%
                
                // 根据宽度和高度计算合适的格子大小
                const widthBasedSize = availableWidth / (maxGridWidth + 2);
                const heightBasedSize = availableHeight / (maxGridHeight + 2);
                
                // 取两者中较小的值，确保棋盘完全可见
                baseCellSize = Math.max(18, Math.min(28, Math.min(widthBasedSize, heightBasedSize)));
            }
            
            gameState.cellSize = baseCellSize;
            
            // 计算当前关卡的画布大小（考虑高DPI）
            const canvasWidth = (gameState.gridSize.width + 1) * gameState.cellSize;
            const canvasHeight = (gameState.gridSize.height + 1) * gameState.cellSize;
            
            // 设置画布CSS尺寸
            canvas.style.width = canvasWidth + 'px';
            canvas.style.height = canvasHeight + 'px';
            
            // 设置画布实际像素尺寸（考虑高DPI）
            const scaledWidth = canvasWidth * gameState.dpr;
            const scaledHeight = canvasHeight * gameState.dpr;
            
            canvas.width = scaledWidth;
            canvas.height = scaledHeight;
            
            // 缩放上下文以匹配高DPI
            ctx.scale(gameState.dpr, gameState.dpr);
            
            // 重新绘制
            drawEverything();
        }

        // 切换暂停/继续状态
        function togglePauseContinue() {
            if (gameState.isPaused) {
                // 继续作答
                continueAnswering();
            } else {
                // 暂停作答
                pauseAnswering();
            }
        }

        // 暂停作答
        function pauseAnswering() {
            if (!gameState.isAnswering || gameState.isCompleted || gameState.isPaused) return;
            
            gameState.isPaused = true;
            gameState.pauseStartTime = Date.now();
            pauseContinueBtn.textContent = '继续作答';
            
            // 显示暂停提示
            pauseOverlay.style.display = 'flex';
            
            // 停止计时器
            if (gameState.timerInterval) {
                clearInterval(gameState.timerInterval);
                gameState.timerInterval = null;
                gameState.timerRunning = false;
            }
            
            // 禁用除了退出、清空、重新开始、继续作答按钮外的所有按钮
            disableButtonsInPauseMode();
            
            showMessage("暂停", "游戏已暂停，计时器已停止。");
        }

        // 继续作答
        function continueAnswering() {
            if (!gameState.isPaused) return;
            
            gameState.isPaused = false;
            
            // 隐藏暂停提示
            pauseOverlay.style.display = 'none';
            
            // 计算暂停时间
            const pauseEndTime = Date.now();
            gameState.pausedDuration += (pauseEndTime - gameState.pauseStartTime);
            
            pauseContinueBtn.textContent = '暂停做答';
            
            // 重新开始计时
            if (gameState.isAnswering && !gameState.isCompleted) {
                gameState.timerRunning = true;
                gameState.timerInterval = setInterval(() => {
                    gameState.elapsedTime = Date.now() - gameState.startTime - gameState.pausedDuration;
                    updateTimerDisplay();
                }, 10);
            }
            
            // 启用所有按钮
            enableButtonsAfterPause();
            
            closeMessage();
        }

        // 在暂停模式下禁用按钮
        function disableButtonsInPauseMode() {
            // 禁用控制按钮（除了退出、清空、重新开始、继续作答）
            document.querySelectorAll('.control-btn').forEach(btn => {
                if (btn.id !== 'clearAllPointsBtn' && btn.id !== 'restartGameBtn' && btn.id !== 'pauseContinueBtn') {
                    btn.disabled = true;
                }
            });
            
            // 禁用工具按钮
            document.querySelectorAll('.tool-btn').forEach(btn => {
                btn.disabled = true;
            });
            
            // 禁用特殊按钮
            specialActionBtn.disabled = true;
            
            // 禁用标注类型按钮
            document.querySelectorAll('.tool-annotation-type').forEach(btn => {
                btn.disabled = true;
            });
        }

            if (gameState.isCompleted) {
                document.querySelectorAll('.control-btn:not(#restartGameBtn)').forEach(btn => {
                    btn.disabled = true;
                });
                
                document.querySelectorAll('.tool-btn').forEach(btn => {
                    btn.disabled = true;
                });
                
                specialActionBtn.disabled = true;
            }
        }

        // 根据阶段更新按钮文本
        function updateButtonText() {
            if (gameState.phase === 'placing') {
                clearAllPointsBtn.textContent = '清除所有点/墙';
            } else {
                clearAllPointsBtn.textContent = '清除所有线段';
            }
        }

        // 处理画布点击
        function handleCanvasClick(event) {
            if (gameState.answerLocked || gameState.isPaused) return;
            
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / canvas.clientWidth;
            const scaleY = canvas.height / canvas.clientHeight;
            
            // 考虑高DPI缩放
            const x = (event.clientX - rect.left) * scaleX / gameState.dpr;
            const y = (event.clientY - rect.top) * scaleY / gameState.dpr;
            
            // 将点击坐标转换为网格坐标（放宽判定范围）
            const gridPos = screenToGrid(x, y);
            
            if (gridPos) {
                if (gameState.phase === 'placing') {
                    handlePlacingPhase(gridPos.x, gridPos.y);
                } else if (gameState.phase === 'connecting') {
                    // 第二阶段只有在开始答题后才能操作
                    if (!gameState.isAnswering && gameState.currentMode === 'line') {
                        showMessage("提示", "请先点击'开始做答'按钮才能开始连线！");
                        return;
                    }
                    
                    if (gameState.currentMode === 'line') {
                        handleConnectingPhase(gridPos.x, gridPos.y);
                    } else {
                        handleAnnotationPhase(gridPos.x, gridPos.y);
                    }
                }
            }
        }

        // 处理触摸事件
        function handleTouch(event) {
            event.preventDefault();
            const touch = event.touches[0];
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / canvas.clientWidth;
            const scaleY = canvas.height / canvas.clientHeight;
            
            // 考虑高DPI缩放
            const x = (touch.clientX - rect.left) * scaleX / gameState.dpr;
            const y = (touch.clientY - rect.top) * scaleY / gameState.dpr;
            
            // 将触摸坐标转换为网格坐标（放宽判定范围）
            const gridPos = screenToGrid(x, y);
            
            if (gridPos) {
                if (gameState.phase === 'placing') {
                    handlePlacingPhase(gridPos.x, gridPos.y);
                } else if (gameState.phase === 'connecting') {
                    // 第二阶段只有在开始答题后才能操作
                    if (!gameState.isAnswering && gameState.currentMode === 'line') {
                        showMessage("提示", "请先点击'开始做答'按钮才能开始连线！");
                        return;
                    }
                    
                    if (gameState.currentMode === 'line') {
                        handleConnectingPhase(gridPos.x, gridPos.y);
                    } else {
                        handleAnnotationPhase(gridPos.x, gridPos.y);
                    }
                }
            }
        }

        // 将屏幕坐标转换为网格坐标（放宽判定范围）
        function screenToGrid(screenX, screenY) {
            // 计算网格起始位置
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            
            // 计算点击位置到最近网格点的距离
            const gridX = (screenX - gridStartX) / gameState.cellSize;
            const gridY = (screenY - gridStartY) / gameState.cellSize;
            
            // 计算最近的网格点（使用四舍五入，增加容错）
            const nearestX = Math.round(gridX);
            const nearestY = Math.round(gridY);
            
            // 检查距离是否在容错范围内（0.6个格子，进一步放宽）
            const distance = Math.sqrt(
                Math.pow(gridX - nearestX, 2) + 
                Math.pow(gridY - nearestY, 2)
            );
            
            if (distance <= 0.6 && 
                nearestX >= 0 && nearestX <= gameState.gridSize.width && 
                nearestY >= 0 && nearestY <= gameState.gridSize.height) {
                return { x: nearestX, y: nearestY };
            }
            
            return null;
        }

        // 处理放置阶段
        function handlePlacingPhase(x, y) {
            // 检查是否在墙线上
            if (gameState.currentTool === 'point' && isPointOnWallLine(x, y)) {
                showMessage("提示", "该位置有墙线，请先移除墙线或选择其他位置！");
                return;
            }
            
            if (gameState.currentTool === 'point') {
                // 检查该位置是否已有点或墙
                const existingPoint = gameState.points.find(p => p.x === x && p.y === y);
                const existingWall = gameState.walls.find(w => w.x === x && w.y === y);
                
                if (existingWall) {
                    showMessage("提示", "该位置已有墙，请先移除墙或选择其他位置！");
                    return;
                }
                
                if (existingPoint) {
                    // 如果已有点，则移除它
                    removePoint(existingPoint);
                } else {
                    // 添加新点
                    addPoint(x, y);
                }
            } else if (gameState.currentTool === 'wall') {
                // 放置墙点
                // 检查该位置是否已有点或墙
                const existingPoint = gameState.points.find(p => p.x === x && p.y === y);
                const existingWall = gameState.walls.find(w => w.x === x && w.y === y);
                
                if (existingPoint) {
                    showMessage("提示", "该位置已有点，请先移除点或选择其他位置！");
                    return;
                }
                
                if (existingWall) {
                    // 如果已有墙点，则移除它
                    removeWall(existingWall);
                } else {
                    // 添加新墙点
                    addWall(x, y);
                }
            } else {
                // 放置墙线
                handleWallLinePhase(x, y);
            }
            
            saveToHistory();
            updateUI();
            drawEverything();
        }

        // 检查点是否在墙线上
        function isPointOnWallLine(x, y) {
            const pointKey = `${x},${y}`;
            return gameState.wallLinePoints.has(pointKey);
        }

        // 处理墙线放置
        function handleWallLinePhase(x, y) {
            if (!gameState.selectedWallLineStart) {
                // 如果没有已选起点，则检查该位置是否可以作为起点
                // 墙线可以从任意位置开始
                gameState.selectedWallLineStart = { x, y };
            } else {
                // 如果已有点被选中，尝试绘制或删除墙线
                if (gameState.selectedWallLineStart.x !== x || gameState.selectedWallLineStart.y !== y) {
                    // 检查是否已经存在这条墙线
                    const existingLineIndex = findWallLineIndex(gameState.selectedWallLineStart, {x, y});
                    
                    if (existingLineIndex !== -1) {
                        // 如果已存在，则删除这条墙线
                        gameState.wallLines.splice(existingLineIndex, 1);
                        updateWallLinePoints();
                        gameState.selectedWallLineStart = null;
                    } else {
                        // 否则尝试添加新墙线
                        const lineAdded = addWallLine(gameState.selectedWallLineStart, {x, y});
                        
                        if (lineAdded) {
                            // 如果成功添加墙线，将当前点设为新的选中点
                            gameState.selectedWallLineStart = {x, y};
                        } else {
                            // 如果无法添加墙线，取消选择
                            gameState.selectedWallLineStart = null;
                        }
                    }
                } else {
                    // 点击同一个点，取消选择
                    gameState.selectedWallLineStart = null;
                }
            }
        }

        // 处理连线阶段
        function handleConnectingPhase(x, y) {
            // 查找点击位置的点
            const clickedPoint = gameState.points.find(p => p.x === x && p.y === y);
            
            if (!clickedPoint) {
                // 如果没有点到点，则取消选择
                gameState.selectedPoint = null;
                drawEverything();
                return;
            }
            
            if (!gameState.selectedPoint) {
                // 如果没有已选点，则选择当前点
                gameState.selectedPoint = clickedPoint;
            } else {
                // 如果已有点被选中，尝试绘制或删除直线
                if (gameState.selectedPoint !== clickedPoint) {
                    // 检查是否已经存在这条线
                    const existingLineIndex = findLineIndex(gameState.selectedPoint, clickedPoint);
                    
                    if (existingLineIndex !== -1) {
                        // 如果已存在，则删除这条线
                        gameState.lines.splice(existingLineIndex, 1);
                        gameState.selectedPoint = null;
                    } else {
                        // 否则尝试添加新直线
                        const lineAdded = addLine(gameState.selectedPoint, clickedPoint);
                        
                        if (lineAdded) {
                            // 如果成功添加直线，将当前点设为新的选中点
                            gameState.selectedPoint = clickedPoint;
                        } else {
                            // 如果无法添加直线，取消选择
                            gameState.selectedPoint = null;
                        }
                    }
                } else {
                    // 点击同一个点，取消选择
                    gameState.selectedPoint = null;
                }
            }
            
            saveToHistory();
            updateUI();
            drawEverything();
        }

        // 处理标注阶段
        function handleAnnotationPhase(x, y) {
            // 查找点击位置的点
            const clickedPoint = gameState.points.find(p => p.x === x && p.y === y);
            
            if (!clickedPoint) {
                return;
            }
            
            // 设置或移除标注
            if (gameState.currentAnnotation === 'none') {
                gameState.annotations.delete(clickedPoint.id);
            } else {
                gameState.annotations.set(clickedPoint.id, gameState.currentAnnotation);
            }
            
            saveToHistory();
            drawEverything();
        }

        // 查找墙线索引
        function findWallLineIndex(point1, point2) {
            return gameState.wallLines.findIndex(line => 
                (line.start.x === point1.x && line.start.y === point1.y && 
                 line.end.x === point2.x && line.end.y === point2.y) ||
                (line.start.x === point2.x && line.start.y === point2.y && 
                 line.end.x === point1.x && line.end.y === point1.y)
            );
        }

        // 查找直线索引
        function findLineIndex(point1, point2) {
            return gameState.lines.findIndex(line => 
                (line.start.x === point1.x && line.start.y === point1.y && 
                 line.end.x === point2.x && line.end.y === point2.y) ||
                (line.start.x === point2.x && line.start.y === point2.y && 
                 line.end.x === point1.x && line.end.y === point1.y)
            );
        }

        // 添加点
        function addPoint(x, y) {
            gameState.points.push({
                x, y,
                id: gameState.pointIdCounter++
            });
        }

        // 移除点
        function removePoint(point) {
            // 移除该点
            const pointIndex = gameState.points.indexOf(point);
            if (pointIndex !== -1) {
                gameState.points.splice(pointIndex, 1);
            }
            
            // 移除该点的标注
            gameState.annotations.delete(point.id);
            
            // 移除所有连接到该点的直线
            gameState.lines = gameState.lines.filter(line => 
                !(line.start.x === point.x && line.start.y === point.y) &&
                !(line.end.x === point.x && line.end.y === point.y)
            );
        }

        // 添加墙点
        function addWall(x, y) {
            gameState.walls.push({
                x, y,
                id: gameState.wallIdCounter++
            });
        }

        // 移除墙点
        function removeWall(wall) {
            const wallIndex = gameState.walls.indexOf(wall);
            if (wallIndex !== -1) {
                gameState.walls.splice(wallIndex, 1);
            }
        }

        // 更新墙线经过的点
        function updateWallLinePoints() {
            gameState.wallLinePoints.clear();
            gameState.wallLines.forEach(line => {
                const dx = line.end.x - line.start.x;
                const dy = line.end.y - line.start.y;
                const steps = Math.max(Math.abs(dx), Math.abs(dy));
                
                for (let i = 0; i <= steps; i++) {
                    const x = line.start.x + Math.round(dx * i / steps);
                    const y = line.start.y + Math.round(dy * i / steps);
                    gameState.wallLinePoints.add(`${x},${y}`);
                }
            });
        }

        // 检查是否与墙点或墙线冲突
        function checkWallConflict(point1, point2) {
            const dx = point2.x - point1.x;
            const dy = point2.y - point1.y;
            const steps = Math.max(Math.abs(dx), Math.abs(dy));
            
            for (let i = 0; i <= steps; i++) {
                const x = point1.x + Math.round(dx * i / steps);
                const y = point1.y + Math.round(dy * i / steps);
                
                // 检查点周围1/4格边长范围内是否有墙点
                for (const wall of gameState.walls) {
                    const wallDx = Math.abs(wall.x - x);
                    const wallDy = Math.abs(wall.y - y);
                    
                    // 使用曼哈顿距离检查1/4格范围
                    if (wallDx <= 0.25 && wallDy <= 0.25) {
                        return true;
                    }
                }
                
                // 检查墙线
                for (const wallLine of gameState.wallLines) {
                    // 检查线段与墙线是否接近
                    const distance = pointToLineDistance(x, y, wallLine.start, wallLine.end);
                    if (distance <= 0.25) {
                        return true;
                    }
                }
            }
            
            return false;
        }

        // 计算点到线段的距离
        function pointToLineDistance(px, py, lineStart, lineEnd) {
            const lineDx = lineEnd.x - lineStart.x;
            const lineDy = lineEnd.y - lineStart.y;
            const lineLength = Math.sqrt(lineDx * lineDx + lineDy * lineDy);
            
            if (lineLength === 0) {
                // 线段退化为点
                const dx = px - lineStart.x;
                const dy = py - lineStart.y;
                return Math.sqrt(dx * dx + dy * dy);
            }
            
            // 计算投影点
            const t = Math.max(0, Math.min(1, ((px - lineStart.x) * lineDx + (py - lineStart.y) * lineDy) / (lineLength * lineLength)));
            const projectionX = lineStart.x + t * lineDx;
            const projectionY = lineStart.y + t * lineDy;
            
            const dx = px - projectionX;
            const dy = py - projectionY;
            return Math.sqrt(dx * dx + dy * dy);
        }

        // 添加墙线
        function addWallLine(point1, point2) {
            // 计算角度
            const dx = point2.x - point1.x;
            const dy = point2.y - point1.y;
            
            // 检查是否为有效角度（0°, 45°, 90°, 135°）
            let angle = null;
            
            if (dx === 0 && dy === 0) {
                return false; // 同一点，不绘制
            } else if (dx === 0) {
                angle = 90; // 垂直线
            } else if (dy === 0) {
                angle = 0; // 水平线
            } else if (Math.abs(dx) === Math.abs(dy)) {
                angle = dx * dy > 0 ? 45 : 135; // 对角线
            } else {
                return false; // 不是有效角度
            }
            
            // 检查直线是否穿过其他点
            if (!isLineClear(point1, point2, true)) {
                return false;
            }
            
            // 添加新墙线
            gameState.wallLines.push({
                start: { x: point1.x, y: point1.y },
                end: { x: point2.x, y: point2.y },
                angle: angle
            });
            
            // 更新墙线经过的点
            updateWallLinePoints();
            
            return true;
        }

        // 添加直线
        function addLine(point1, point2) {
            // 计算角度
            const dx = point2.x - point1.x;
            const dy = point2.y - point1.y;
            
            // 检查是否为有效角度（0°, 45°, 90°, 135°）
            let angle = null;
            
            if (dx === 0 && dy === 0) {
                return false; // 同一点，不绘制
            } else if (dx === 0) {
                angle = 90; // 垂直线
            } else if (dy === 0) {
                angle = 0; // 水平线
            } else if (Math.abs(dx) === Math.abs(dy)) {
                angle = dx * dy > 0 ? 45 : 135; // 对角线
            } else {
                return false; // 不是有效角度
            }
            
            // 检查直线是否穿过其他点
            if (!isLineClear(point1, point2, false)) {
                return false;
            }
            
            // 检查是否与墙点或墙线冲突
            if (checkWallConflict(point1, point2)) {
                showMessage("错误", "线段不许经过墙的四分之一格边长范围！");
                return false;
            }
            
            // 添加新直线（不再检查连接数限制）
            gameState.lines.push({
                start: { x: point1.x, y: point1.y },
                end: { x: point2.x, y: point2.y },
                angle: angle
            });
            
            return true;
        }

        // 检查直线是否穿过其他点
        function isLineClear(point1, point2, isWallLine = false) {
            const dx = point2.x - point1.x;
            const dy = point2.y - point1.y;
            
            // 计算步长
            const steps = Math.max(Math.abs(dx), Math.abs(dy));
            
            for (let i = 1; i < steps; i++) {
                const x = point1.x + Math.round(dx * i / steps);
                const y = point1.y + Math.round(dy * i / steps);
                
                // 检查中间点是否有点（但不是起点或终点）
                if (!isWallLine) {
                    // 对于普通直线，检查是否穿过其他点
                    if (gameState.points.some(p => p.x === x && p.y === y && 
                        !(p.x === point1.x && p.y === point1.y) && 
                        !(p.x === point2.x && p.y === point2.y))) {
                        return false;
                    }
                }
                
                // 检查是否穿过墙点
                if (gameState.walls.some(w => w.x === x && w.y === y)) {
                    return false;
                }
            }
            
            return true;
        }

        // 检测环
        function detectCycle() {
            if (gameState.points.length === 0) return { hasCycle: false, isComplete: false };
            
            // 构建邻接表
            const adjList = new Map();
            gameState.points.forEach(point => {
                adjList.set(point.id, []);
            });
            
            // 填充邻接表
            gameState.lines.forEach(line => {
                const startPoint = gameState.points.find(p => 
                    p.x === line.start.x && p.y === line.start.y);
                const endPoint = gameState.points.find(p => 
                    p.x === line.end.x && p.y === line.end.y);
                
                if (startPoint && endPoint) {
                    adjList.get(startPoint.id).push(endPoint.id);
                    adjList.get(endPoint.id).push(startPoint.id);
                }
            });
            
            // DFS检测环
            const visited = new Set();
            
            for (const point of gameState.points) {
                if (!visited.has(point.id)) {
                    const hasCycle = dfsDetectCycle(point.id, -1, visited, adjList);
                    if (hasCycle) {
                        // 检查是否访问了所有点
                        const allVisited = gameState.points.every(p => visited.has(p.id));
                        return { hasCycle: true, isComplete: allVisited };
                    }
                }
            }
            
            return { hasCycle: false, isComplete: false };
        }

        // DFS检测环
        function dfsDetectCycle(node, parent, visited, adjList) {
            visited.add(node);
            
            for (const neighbor of adjList.get(node) || []) {
                if (!visited.has(neighbor)) {
                    if (dfsDetectCycle(neighbor, node, visited, adjList)) {
                        return true;
                    }
                } else if (neighbor !== parent) {
                    return true;
                }
            }
            
            return false;
        }

        // 确认完成放置
        function confirmFinishPlacing() {
            if (gameState.points.length < 2) {
                showMessage("提示", "请至少放置2个点再进入连线阶段！");
                return;
            }
            
            showConfirmMessage(
                "确认完成出题",
                "确定已经完成出题，要开始连线吗？",
                () => {
                    setPhase('connecting');
                }
            );
        }

        // 开始答题
        function startAnswering() {
            gameState.isAnswering = true;
            gameState.startTime = Date.now();
            gameState.timerRunning = true;
            gameState.isPaused = false;
            gameState.pausedDuration = 0;
            
            // 确保暂停提示隐藏
            pauseOverlay.style.display = 'none';
            
            // 开始计时（毫秒级）
            if (gameState.timerInterval) {
                clearInterval(gameState.timerInterval);
            }
            
            gameState.timerInterval = setInterval(() => {
                if (!gameState.isPaused) {
                    gameState.elapsedTime = Date.now() - gameState.startTime - gameState.pausedDuration;
                    updateTimerDisplay();
                }
            }, 10); // 每10毫秒更新一次
            
            // 更新按钮
            updateSpecialButton();
            pauseContinueBtn.style.display = 'inline-block';
        }

        // 更新计时器显示（毫秒级）
        function updateTimerDisplay() {
            const totalMilliseconds = gameState.elapsedTime;
            const minutes = Math.floor(totalMilliseconds / 60000);
            const seconds = Math.floor((totalMilliseconds % 60000) / 1000);
            const milliseconds = Math.floor((totalMilliseconds % 1000) / 10);
            
            timerValue.textContent = 
                `${minutes.toString().padStart(2, '0')}:` +
                `${seconds.toString().padStart(2, '0')}.` +
                `${milliseconds.toString().padStart(2, '0')}`;
        }

        // 检查答案
        function checkSolution() {
            // 检查是否有超过2条线连接的点
            const connectionCounts = gameState.points.map(point => {
                return gameState.lines.filter(line => 
                    (line.start.x === point.x && line.start.y === point.y) ||
                    (line.end.x === point.x && line.end.y === point.y)
                ).length;
            });
            
            const overConnectionPoints = connectionCounts.filter(count => count > 2).length;
            
            if (overConnectionPoints > 0) {
                showMessage("错误", "有的点线连多了！");
                // 继续计时，不停止
                return;
            }
            
            // 检查是否有未连接的点
            const zeroConnectionPoints = connectionCounts.filter(count => count === 0).length;
            
            if (zeroConnectionPoints > 0) {
                showMessage("需要继续作答", `还有${zeroConnectionPoints}个点没有连接任何线！`);
                // 继续计时，不停止
                return;
            }
            
            // 检查连接数限制
            const singleConnectionPoints = connectionCounts.filter(count => count === 1).length;
            
            if (singleConnectionPoints > 2) {
                showMessage("需要继续作答", `有${singleConnectionPoints}个点只连接了1条线，最多只能有2个这样的点！`);
                // 继续计时，不停止
                return;
            }
            
            // 检测环
            const cycleResult = detectCycle();
            
            if (cycleResult.hasCycle) {
                if (cycleResult.isComplete) {
                    // 环包含了所有点，成功
                    completeGame();
                } else {
                    showMessage("错误", "成环了，但这个环没有经过所有的点！");
                    // 继续计时，不停止
                    return;
                }
            } else {
                // 没有环，检查是否所有点都连接正确
                if (singleConnectionPoints <= 2) {
                    completeGame();
                } else {
                    showMessage("需要继续作答", "请确保所有点都正确连接！");
                    // 继续计时，不停止
                    return;
                }
            }
        }

        // 完成游戏
        function completeGame() {
            // 停止计时
            if (gameState.timerInterval) {
                clearInterval(gameState.timerInterval);
                gameState.timerInterval = null;
                gameState.timerRunning = false;
            }
            
            // 格式化时间显示
            const totalMilliseconds = gameState.elapsedTime;
            const minutes = Math.floor(totalMilliseconds / 60000);
            const seconds = Math.floor((totalMilliseconds % 60000) / 1000);
            const milliseconds = Math.floor((totalMilliseconds % 1000) / 10);
            const timeString = 
                `${minutes.toString().padStart(2, '0')}:` +
                `${seconds.toString().padStart(2, '0')}.` +
                `${milliseconds.toString().padStart(2, '0')}`;
            
            // 计算钴系数：玩家用时(分钟)/点数*200，四舍五入到个位
            // 注意：elapsedTime是毫秒，除以60000得到分钟
            const cobaltCoefficient = Math.round((gameState.elapsedTime / 60000) / gameState.points.length * 200);
            
            gameState.isCompleted = true;
            gameState.answerLocked = true;
            gameState.isPaused = false;
            
            // 激活特殊烟花按钮
            gameState.specialFireworkActive = true;
            
            // 更新暂停按钮文本
            pauseContinueBtn.textContent = '暂停做答';
            
            // 禁用所有按钮（除了重新开始、退出和返回第一阶段）
            document.querySelectorAll('.control-btn:not(#restartGameBtn)').forEach(btn => {
                btn.disabled = true;
            });
            
            document.querySelectorAll('.tool-btn').forEach(btn => {
                btn.disabled = true;
            });
            
            // 启用返回第一阶段按钮
            returnPhase1Btn.disabled = false;
            
            // 更新特殊按钮为烟花按钮
            updateSpecialButton();
            
            // 显示包含钴系数的成功消息
            showMessageWithSuccess(
                "恭喜！", 
                "恭喜你！做出来了！答案正确！",
                `用时: ${timeString}`,
                `钴系数: ${cobaltCoefficient}`
            );
        }

        // 清除所有点/墙/线段
        function clearAllPoints() {
            if (gameState.phase === 'placing') {
                // 第一阶段：清除所有点、墙点和墙线
                if (gameState.points.length === 0 && gameState.walls.length === 0 && gameState.wallLines.length === 0) {
                    return;
                }
                
                if (confirm("确定要清除所有点、墙点和墙线吗？")) {
                    gameState.points = [];
                    gameState.walls = [];
                    gameState.wallLines = [];
                    gameState.wallLinePoints.clear();
                    gameState.lines = [];
                    gameState.selectedPoint = null;
                    gameState.selectedWallLineStart = null;
                    gameState.pointIdCounter = 1;
                    gameState.wallIdCounter = 1;
                    gameState.annotations.clear();
                    saveToHistory();
                    updateUI();
                    drawEverything();
                }
            } else {
                // 第二阶段：清除所有线段
                if (gameState.lines.length === 0) {
                    showMessage("提示", "已经没有可以删的线段了！！！");
                    return;
                }
                
                if (confirm("确定要清除所有线段吗？")) {
                    gameState.lines = [];
                    gameState.selectedPoint = null;
                    saveToHistory();
                    updateUI();
                    drawEverything();
                }
            }
        }

        // 重新开始游戏
        function restartGame() {
            if (confirm("确定要重新开始游戏吗？所有进度将会丢失。")) {
                // 重置到当前大小的初始状态
                resetGameState();
                setPhase('placing');
                saveToHistory();
                updateUI();
                drawEverything();
            }
        }

        // 撤销上一步操作
        function undoLastAction() {
            if (gameState.phase === 'connecting' && gameState.lines.length === 0 && gameState.currentMode === 'line') {
                showMessage("提示", "已经没有可以删的线段了！！！");
                return;
            }
            
            if (gameState.history.length <= 1) {
                return; // 只有初始状态，无法撤销
            }
            
            // 移除当前状态
            gameState.history.pop();
            
            // 恢复到上一个状态
            const prevState = gameState.history[gameState.history.length - 1];
            gameState.points = [...prevState.points];
            gameState.walls = [...prevState.walls];
            gameState.wallLines = prevState.wallLines ? [...prevState.wallLines] : [];
            gameState.wallLinePoints = prevState.wallLinePoints ? new Set(prevState.wallLinePoints) : new Set();
            gameState.lines = [...prevState.lines];
            gameState.selectedPoint = null;
            gameState.selectedWallLineStart = null;
            gameState.pointIdCounter = prevState.pointIdCounter;
            gameState.wallIdCounter = prevState.wallIdCounter;
            gameState.annotations = new Map(prevState.annotations);
            
            updateUI();
            drawEverything();
        }

        // 保存到历史记录
        function saveToHistory() {
            gameState.history.push({
                points: [...gameState.points],
                walls: [...gameState.walls],
                wallLines: [...gameState.wallLines],
                wallLinePoints: new Set(gameState.wallLinePoints),
                lines: [...gameState.lines],
                pointIdCounter: gameState.pointIdCounter,
                wallIdCounter: gameState.wallIdCounter,
                annotations: new Map(gameState.annotations)
            });
            
            // 限制历史记录数量
            if (gameState.history.length > 20) {
                gameState.history.shift();
            }
        }

        // 绘制网格 - 改回原来的格子颜色
        function drawGrid() {
            ctx.clearRect(0, 0, canvas.width / gameState.dpr, canvas.height / gameState.dpr);
            
            // 设置网格样式 - 改回原来的颜色
            ctx.strokeStyle = '#D2B48C';
            ctx.lineWidth = 1;
            
            // 计算网格起始位置
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            const gridWidth = gameState.gridSize.width * gameState.cellSize;
            const gridHeight = gameState.gridSize.height * gameState.cellSize;
            
            // 绘制网格线
            for (let i = 0; i <= gameState.gridSize.width; i++) {
                // 垂直线
                ctx.beginPath();
                ctx.moveTo(gridStartX + i * gameState.cellSize, gridStartY);
                ctx.lineTo(gridStartX + i * gameState.cellSize, gridStartY + gridHeight);
                ctx.stroke();
            }
            
            for (let i = 0; i <= gameState.gridSize.height; i++) {
                // 水平线
                ctx.beginPath();
                ctx.moveTo(gridStartX, gridStartY + i * gameState.cellSize);
                ctx.lineTo(gridStartX + gridWidth, gridStartY + i * gameState.cellSize);
                ctx.stroke();
            }
            
            // 绘制网格标签（坐标）
            ctx.fillStyle = '#8B5A2B';
            ctx.font = '12px Arial';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            
            // 只在边界显示坐标，避免过于拥挤
            const step = Math.max(1, Math.floor(gameState.gridSize.width / 10));
            for (let i = 0; i <= gameState.gridSize.width; i += step) {
                const x = gridStartX + i * gameState.cellSize;
                // 顶部坐标
                ctx.fillText(i, x, gridStartY - 10);
                // 底部坐标
                ctx.fillText(i, x, gridStartY + gridHeight + 15);
            }
            
            for (let i = 0; i <= gameState.gridSize.height; i += step) {
                const y = gridStartY + i * gameState.cellSize;
                // 左侧坐标
                ctx.fillText(i, gridStartX - 15, y);
                // 右侧坐标
                ctx.fillText(i, gridStartX + gridWidth + 15, y);
            }
        }

        // 绘制墙点
        function drawWalls() {
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            
            ctx.save();
            gameState.walls.forEach(wall => {
                const x = gridStartX + wall.x * gameState.cellSize;
                const y = gridStartY + wall.y * gameState.cellSize;
                const wallSize = gameState.cellSize / 2; // 墙的大小为1/2格边长
                
                // 绘制墙点（灰蓝色大十字）
                ctx.strokeStyle = '#4682B4';
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(x - wallSize/2, y);
                ctx.lineTo(x + wallSize/2, y);
                ctx.moveTo(x, y - wallSize/2);
                ctx.lineTo(x, y + wallSize/2);
                ctx.stroke();
            });
            ctx.restore();
        }

        // 绘制墙线和墙线点
        function drawWallLines() {
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            
            // 先绘制墙线
            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            ctx.strokeStyle = '#87CEEB'; // 天蓝色
            
            gameState.wallLines.forEach(line => {
                // 计算墙线起点和终点的屏幕坐标
                const startX = gridStartX + line.start.x * gameState.cellSize;
                const startY = gridStartX + line.start.y * gameState.cellSize;
                const endX = gridStartX + line.end.x * gameState.cellSize;
                const endY = gridStartX + line.end.y * gameState.cellSize;
                
                // 绘制墙线
                ctx.beginPath();
                ctx.moveTo(startX, startY);
                ctx.lineTo(endX, endY);
                ctx.stroke();
            });
            
            // 绘制墙线经过的点（灰蓝色圆点，直径与墙线宽度一致）
            ctx.fillStyle = '#A3D8F4'; // 灰蓝色
            gameState.wallLinePoints.forEach(pointKey => {
                const [x, y] = pointKey.split(',').map(Number);
                const screenX = gridStartX + x * gameState.cellSize;
                const screenY = gridStartY + y * gameState.cellSize;
                const pointRadius = 3; // 墙线宽度
                
                ctx.beginPath();
                ctx.arc(screenX, screenY, pointRadius, 0, Math.PI * 2);
                ctx.fill();
            });
            
            // 绘制选中的墙线起点（黄色）
            if (gameState.selectedWallLineStart) {
                const x = gridStartX + gameState.selectedWallLineStart.x * gameState.cellSize;
                const y = gridStartY + gameState.selectedWallLineStart.y * gameState.cellSize;
                const pointRadius = gameState.cellSize / 8;
                
                ctx.fillStyle = '#FFD734'; // 黄色
                ctx.beginPath();
                ctx.arc(x, y, pointRadius, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // 绘制点
        function drawPoints() {
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            
            // 先绘制圆圈标注（在点下方）
            drawCircleAnnotations();
            
            // 绘制点
            gameState.points.forEach(point => {
                // 计算点的屏幕坐标
                const x = gridStartX + point.x * gameState.cellSize;
                const y = gridStartY + point.y * gameState.cellSize;
                
                // 检查是否有标注
                const hasAnnotation = gameState.annotations.has(point.id);
                const annotationType = gameState.annotations.get(point.id);
                
                // 如果有非圆圈标注，隐藏原始点
                if (hasAnnotation && annotationType !== 'circle') {
                    return; // 不绘制原始点
                }
                
                // 点的大小：如果有圆圈标注，大小为正常一半；否则正常大小
                const pointRadius = (hasAnnotation && annotationType === 'circle') ? 
                    gameState.cellSize / 8 : // 有圆圈标注时大小为正常一半（1/4格边长的一半）
                    gameState.cellSize / 4;  // 正常大小为1/2格边长
                
                // 绘制点
                ctx.beginPath();
                ctx.arc(x, y, pointRadius, 0, Math.PI * 2);
                
                // 如果点被选中，使用红色，否则使用黑色
                if (gameState.selectedPoint === point) {
                    ctx.fillStyle = '#FF6B6B'; // 选中状态 - 红色
                } else {
                    ctx.fillStyle = '#000000'; // 正常状态 - 黑色
                }
                
                ctx.fill();
                
                // 绘制点边框 - 与普通点尺寸相同的白边
                ctx.lineWidth = 1;
                ctx.strokeStyle = '#fff';
                ctx.stroke();
            });
            
            // 绘制非圆圈标注（在点上方）
            drawNonCircleAnnotations();
        }

        // 在点下方绘制圆圈标注
        function drawCircleAnnotations() {
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            const baseSize = gameState.cellSize / 2; // 基础大小为1/2格边长
            
            // 只绘制圆圈标注（在点下方）
            gameState.annotations.forEach((type, pointId) => {
                if (type !== 'circle') return;
                
                const point = gameState.points.find(p => p.id === pointId);
                if (!point) return;
                
                const x = gridStartX + point.x * gameState.cellSize;
                const y = gridStartY + point.y * gameState.cellSize;
                
                ctx.save();
                
                // 空心圆圈，直径2/3格边长
                ctx.strokeStyle = gameState.selectedPoint && gameState.selectedPoint.id === pointId ? 
                    '#FF6B6B' : '#000000'; // 选中时红色，否则黑色
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.arc(x, y, baseSize * 2/3, 0, Math.PI * 2);
                ctx.stroke();
                
                ctx.restore();
            });
        }

        // 在点上方绘制非圆圈标注 - 增强白边效果
        function drawNonCircleAnnotations() {
            const gridStartX = gameState.canvasPadding;
            const gridStartY = gameState.canvasPadding;
            const baseSize = gameState.cellSize / 2; // 基础大小为1/2格边长
            
            // 绘制除圆圈外的所有标注（在点上方）
            gameState.annotations.forEach((type, pointId) => {
                if (type === 'circle') return; // 圆圈已经在下方绘制
                
                const point = gameState.points.find(p => p.id === pointId);
                if (!point) return;
                
                const x = gridStartX + point.x * gameState.cellSize;
                const y = gridStartY + point.y * gameState.cellSize;
                
                ctx.save();
                
                // 设置颜色：选中时红色，否则黑色
                const color = gameState.selectedPoint && gameState.selectedPoint.id === pointId ? 
                    '#FF6B6B' : '#000000';
                
                // 计算普通点的尺寸（1/2格边长）
                const normalPointSize = gameState.cellSize / 2;
                
                // 增强白边效果 - 增加白边大小
                const whiteBorderSize = normalPointSize * 1.3; // 增加30%的白边大小
                
                // 绘制白色边框 - 增强白边效果
                ctx.fillStyle = '#FFFFFF';
                ctx.strokeStyle = '#FFFFFF';
                ctx.lineWidth = 2;
                
                switch(type) {
                    case 'square':
                        // 绘制白色方形（增强白边）
                        ctx.beginPath();
                        ctx.rect(x - whiteBorderSize/2, y - whiteBorderSize/2, whiteBorderSize, whiteBorderSize);
                        ctx.fill();
                        // 内部填充（稍小一点）
                        ctx.fillStyle = color;
                        ctx.beginPath();
                        ctx.rect(x - baseSize/2, y - baseSize/2, baseSize, baseSize);
                        ctx.fill();
                        break;
                        
                    case 'triangle':
                        // 绘制白色三角形（增强白边）
                        const whiteTriangleHeight = whiteBorderSize;
                        const whiteTriangleHalfBase = whiteBorderSize/2;
                        ctx.beginPath();
                        ctx.moveTo(x, y - whiteTriangleHeight/2);
                        ctx.lineTo(x - whiteTriangleHalfBase, y + whiteTriangleHeight/2);
                        ctx.lineTo(x + whiteTriangleHalfBase, y + whiteTriangleHeight/2);
                        ctx.closePath();
                        ctx.fill();
                        // 内部填充（稍小一点）
                        ctx.fillStyle = color;
                        ctx.beginPath();
                        ctx.moveTo(x, y - baseSize/2);
                        ctx.lineTo(x - baseSize/2, y + baseSize/2);
                        ctx.lineTo(x + baseSize/2, y + baseSize/2);
                        ctx.closePath();
                        ctx.fill();
                        break;
                        
                    case 'diamond':
                        // 绘制白色菱形（增强白边）
                        const whiteDiamondSize = whiteBorderSize;
                        ctx.beginPath();
                        ctx.moveTo(x, y - whiteDiamondSize/2);
                        ctx.lineTo(x + whiteDiamondSize/2, y);
                        ctx.lineTo(x, y + whiteDiamondSize/2);
                        ctx.lineTo(x - whiteDiamondSize/2, y);
                        ctx.closePath();
                        ctx.fill();
                        // 内部填充（稍小一点）
                        ctx.fillStyle = color;
                        ctx.beginPath();
                        ctx.moveTo(x, y - baseSize/2);
                        ctx.lineTo(x + baseSize/2, y);
                        ctx.lineTo(x, y + baseSize/2);
                        ctx.lineTo(x - baseSize/2, y);
                        ctx.closePath();
                        ctx.fill();
                        break;
                        
                    case 'star':
                        // 绘制白色五角星（增强白边，变小一点）
                        const whiteStarOuter = whiteBorderSize/2 * 1.1;
                        const whiteStarInner = whiteBorderSize/4 * 1.1;
                        ctx.beginPath();
                        drawStarShape(ctx, x, y, 5, whiteStarOuter, whiteStarInner);
                        ctx.closePath();
                        ctx.fill();
                        // 内部填充（稍小一点）
                        ctx.fillStyle = color;
                        ctx.beginPath();
                        drawStarShape(ctx, x, y, 5, baseSize/2 * 1.0, baseSize/4 * 1.0);
                        ctx.closePath();
                        ctx.fill();
                        break;
                        
                    case 'cross':
                        // 绘制白色叉（增强白边）
                        const crossSize = whiteBorderSize/2;
                        ctx.lineWidth = 4; // 增加白边线条宽度
                        ctx.beginPath();
                        ctx.moveTo(x - crossSize, y - crossSize);
                        ctx.lineTo(x + crossSize, y + crossSize);
                        ctx.moveTo(x + crossSize, y - crossSize);
                        ctx.lineTo(x - crossSize, y + crossSize);
                        ctx.stroke();
                        
                        // 内部线条（稍小一点）
                        ctx.strokeStyle = color;
                        ctx.lineWidth = 2;
                        ctx.beginPath();
                        ctx.moveTo(x - baseSize/2, y - baseSize/2);
                        ctx.lineTo(x + baseSize/2, y + baseSize/2);
                        ctx.moveTo(x + baseSize/2, y - baseSize/2);
                        ctx.lineTo(x - baseSize/2, y + baseSize/2);
                        ctx.stroke();
                        break;
                        
                    case 'check':
                        // 绘制白色对勾（增强白边，没有实心部分）
                        const checkSize = whiteBorderSize/2;
                        ctx.lineWidth = 4; // 增加白边线条宽度
                        ctx.beginPath();
                        ctx.moveTo(x - checkSize, y);
                        ctx.lineTo(x - checkSize/3, y + checkSize);
                        ctx.lineTo(x + checkSize, y - checkSize);
                        ctx.stroke();
                        
                        // 内部线条（稍小一点）
                        ctx.strokeStyle = color;
                        ctx.lineWidth = 2;
                        ctx.beginPath();
                        ctx.moveTo(x - baseSize/2, y);
                        ctx.lineTo(x - baseSize/6, y + baseSize/2);
                        ctx.lineTo(x + baseSize/2, y - baseSize/2);
                        ctx.stroke();
                        break;
                }
                
                ctx.restore();
            });
        }

        // 绘制五角星形状
        function drawStarShape(ctx, cx, cy, spikes, outerRadius, innerRadius) {
            let rot = Math.PI / 2 * 3;
            let x = cx;
            let y = cy;
            const step = Math.PI / spikes;
            
            ctx.beginPath();
            ctx.moveTo(cx, cy - outerRadius);
            
            for (let i = 0; i < spikes; i++) {
                x = cx + Math.cos(rot) * outerRadius;
                y = cy + Math.sin(rot) * outerRadius;
                ctx.lineTo(x, y);
                rot += step;
                
                x = cx + Math.cos(rot) * innerRadius;
                y = cy + Math.sin(rot) * innerRadius;
                ctx.lineTo(x, y);
                rot += step;
            }
            
            ctx.lineTo(cx, cy - outerRadius);
            ctx.closePath();
        }

        }

        // 更新UI
        function updateUI() {
            // 更新计数
            pointCountSpan.textContent = gameState.points.length;
            lineCountSpan.textContent = gameState.lines.length;
            pointCountValue.textContent = gameState.points.length;
            
            // 计算连接1条线的点的数量
            const singleConnectionPoints = gameState.points.filter(point => {
                const connectionCount = gameState.lines.filter(line => 
                    (line.start.x === point.x && line.start.y === point.y) ||
                    (line.end.x === point.x && line.end.y === point.y)
                ).length;
                return connectionCount === 1;
            }).length;
            
            singleConnectionCountSpan.textContent = singleConnectionPoints;
        }

        // 页面加载完成后初始化游戏
        window.addEventListener('load', initGame);
